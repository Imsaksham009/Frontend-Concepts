## Error Types — Built-in Errors

```javascript
// JavaScript has 7 built-in error types
ReferenceError  // accessing undeclared variable
TypeError       // wrong type operation
SyntaxError     // invalid syntax — caught at parse time
RangeError      // value out of valid range
URIError        // malformed URI functions
EvalError       // eval() related (rare)
Error           // base error type — use for custom errors

// Each has name and message
try {
    null.property;
} catch(err) {
    console.log(err instanceof TypeError); // true
    console.log(err.name);                 // 'TypeError'
    console.log(err.message);              // "Cannot read properties of null"
    console.log(err.stack);               // full stack trace
}
```

## `try/catch/finally` — Complete Mechanics

```javascript
function divide(a, b) {
    try {
        // Code that might throw
        if (typeof a !== 'number') throw new TypeError('a must be a number');
        if (b === 0) throw new RangeError('Cannot divide by zero');
        return a / b;
        
    } catch(err) {
        // Catches ANY error thrown in try block
        // err is the thrown value — can be anything, not just Error objects
        
        if (err instanceof TypeError) {
            console.log('Type problem:', err.message);
            return null;
        }
        
        if (err instanceof RangeError) {
            console.log('Range problem:', err.message);
            return Infinity;
        }
        
        throw err; // re-throw unknown errors — don't swallow them
        
    } finally {
        // ALWAYS runs — whether try succeeded, catch ran, or even if
        // try/catch contains a return statement
        console.log('divide() finished');
        // WARNING: returning from finally OVERRIDES try/catch return
        // Don't return from finally unless you intend to override
    }
}

divide(10, 2);  // returns 5, logs 'divide() finished'
divide('x', 2); // returns null, logs 'divide() finished'
divide(10, 0);  // returns Infinity, logs 'divide() finished'
```

### `finally` Always Runs — Even With Return

```javascript
function test() {
    try {
        return 'from try'; // would return this
    } finally {
        return 'from finally'; // OVERRIDES the try return
    }
}

test(); // 'from finally'

// Practical use — cleanup that must happen regardless
async function fetchData() {
    showLoader();
    try {
        const data = await fetch('/api/data');
        return await data.json();
    } catch(err) {
        console.error(err);
        return null;
    } finally {
        hideLoader(); // always hides loader — success or failure
    }
}
```

## Custom Error Classes

```javascript
// Extend the base Error class
class AppError extends Error {
    constructor(message, code, statusCode = 500) {
        super(message);          // sets this.message
        this.name = 'AppError'; // override default 'Error'
        this.code = code;
        this.statusCode = statusCode;
        
        // Maintains proper stack trace in V8 (Node/Chrome)
        if (Error.captureStackTrace) {
            Error.captureStackTrace(this, AppError);
        }
    }
}

class ValidationError extends AppError {
    constructor(message, fields) {
        super(message, 'VALIDATION_ERROR', 400);
        this.name = 'ValidationError';
        this.fields = fields; // which fields failed validation
    }
}

class NotFoundError extends AppError {
    constructor(resource, id) {
        super(`${resource} with id ${id} not found`, 'NOT_FOUND', 404);
        this.name = 'NotFoundError';
    }
}

class AuthError extends AppError {
    constructor(message = 'Unauthorized') {
        super(message, 'AUTH_ERROR', 401);
        this.name = 'AuthError';
    }
}

// Usage
function getUser(id) {
    const user = db.find(id);
    if (!user) throw new NotFoundError('User', id);
    return user;
}

function createUser(data) {
    const errors = validate(data);
    if (errors.length) {
        throw new ValidationError('Validation failed', errors);
    }
    return db.create(data);
}

// Handle different error types differently
try {
    const user = getUser(99);
} catch(err) {
    if (err instanceof NotFoundError) {
        res.status(404).json({ error: err.message });
    } else if (err instanceof ValidationError) {
        res.status(400).json({ error: err.message, fields: err.fields });
    } else if (err instanceof AuthError) {
        res.status(401).json({ error: err.message });
    } else {
        // Unknown error — log it, don't expose internals
        console.error(err);
        res.status(500).json({ error: 'Internal server error' });
    }
}
```

## Error Propagation

```javascript
// Errors bubble up the call stack until caught
function c() { throw new Error('from c'); }
function b() { c(); }          // doesn't catch — propagates up
function a() { b(); }          // doesn't catch — propagates up

try {
    a();
} catch(err) {
    console.log(err.message); // 'from c' — caught here
}

// Async error propagation — must use async/await or .catch()
async function fetchAndProcess() {
    const data  = await fetch('/api');    // throws on network error
    const json  = await data.json();      // throws on parse error
    return process(json);                  // throws on logic error
    // ALL of these bubble up to the caller
}

async function main() {
    try {
        const result = await fetchAndProcess();
    } catch(err) {
        // Catches errors from any step inside fetchAndProcess
        console.error(err);
    }
}
```

## Global Error Handling — Last Resort

```javascript
// Browser — unhandled errors
window.addEventListener('error', event => {
    console.error('Uncaught error:', event.error);
    // Log to monitoring service
    logToSentry(event.error);
});

// Browser — unhandled Promise rejections
window.addEventListener('unhandledrejection', event => {
    console.error('Unhandled rejection:', event.reason);
    event.preventDefault(); // prevents console warning
    logToSentry(event.reason);
});

// Node.js
process.on('uncaughtException', (err) => {
    console.error('Uncaught exception:', err);
    // Cleanup and exit — don't try to continue
    process.exit(1);
});

process.on('unhandledRejection', (reason, promise) => {
    console.error('Unhandled rejection at:', promise, 'reason:', reason);
});
```

## Error Handling Patterns

### Result Pattern — Avoid Throwing Entirely

```javascript
// Instead of throwing, return a result object
// Common in Rust, Go — becoming popular in JS
function divide(a, b) {
    if (b === 0) {
        return { ok: false, error: new RangeError('Division by zero') };
    }
    return { ok: true, value: a / b };
}

const result = divide(10, 0);

if (result.ok) {
    console.log(result.value);
} else {
    console.log(result.error.message);
}

// With async operations
async function fetchUser(id) {
    try {
        const user = await api.getUser(id);
        return { ok: true, value: user };
    } catch(err) {
        return { ok: false, error: err };
    }
}

const { ok, value: user, error } = await fetchUser(1);
if (!ok) handleError(error);
```

### Error Boundary Pattern

```javascript
// Wrap risky operations with context
function withErrorContext(context, fn) {
    try {
        return fn();
    } catch(err) {
        err.message = `[${context}] ${err.message}`;
        throw err;
    }
}

// Async version
async function withErrorContextAsync(context, fn) {
    try {
        return await fn();
    } catch(err) {
        err.context = context;
        err.message = `[${context}] ${err.message}`;
        throw err;
    }
}

// Usage
const user = withErrorContext('UserService.getUser', () => {
    return db.query(`SELECT * FROM users WHERE id = ${id}`);
});
// Error message: "[UserService.getUser] connection refused"
```

## The Key Rules

```
1. Always handle or re-throw — never silently swallow errors
2. Use specific error types — catch what you can handle
3. Re-throw what you can't handle — let it bubble up
4. finally for cleanup — not for return values
5. Custom error classes — carry context, not just messages
6. Never return from finally — overrides try/catch return
7. Global handlers — last resort logging, not business logic
8. Async errors need try/catch or .catch() — they don't bubble automatically
```
